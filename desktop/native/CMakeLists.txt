cmake_minimum_required(VERSION 3.24)
project(JustWhisperRuntime LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU baseline" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Avoid a separate OpenMP runtime" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "Build CLI recovery" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper
  URL https://github.com/ggml-org/whisper.cpp/archive/refs/tags/v1.9.4.tar.gz
  URL_HASH SHA256=57e280cee375ab02425b806ad5146b99f6eb9357e3c2b31357c8a6af2e2e44ae
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(whisper)
add_executable(justwhisper-live ../../native/live-whisper.cpp)
target_compile_features(justwhisper-live PRIVATE cxx_std_17)
target_link_libraries(justwhisper-live PRIVATE whisper)
install(TARGETS justwhisper-live whisper-cli RUNTIME DESTINATION . COMPONENT JustWhisperRuntime)
install(FILES ${whisper_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME whisper.cpp-LICENSE COMPONENT JustWhisperRuntime)
